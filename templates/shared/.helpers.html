{{/* notify publishes a lightweight ping on the per-user bus topic so other tabs refresh. */}} {{define "notify"}} {{- $owner := or (.Req.Header.Get "X-Token-Subject") "anonymous"}} {{- $_ := .Bus.Publish (printf "todos:%s" $owner) "updated"}} {{end}} {{/* ensure-lists creates a default "My Todos" list when the owner has none, and attaches any orphan todos (list_id NULL) to that list. */}} {{define "ensure-lists"}} {{- $owner := or (.Req.Header.Get "X-Token-Subject") "anonymous"}} {{- $n := .DB.QueryVal `SELECT COUNT(*) FROM todo_list WHERE owner_id = ?` $owner}} {{- if eq $n 0}} {{- $id := uuidv4}} {{- $_ := .DB.Exec `INSERT INTO todo_list (id, owner_id, name) VALUES (?, ?, ?)` $id $owner "My Todos"}} {{- end}} {{- $_ := .DB.Exec ` UPDATE todo SET list_id = ( SELECT id FROM todo_list WHERE owner_id = ? ORDER BY rowid, id LIMIT 1 ) WHERE owner_id = ? AND (list_id IS NULL OR list_id = '')` $owner $owner}} {{end}} {{/* list-path — canonical page URL for a list (trailing slash; file-based index route). list-base — same without trailing slash; use for DELETE and as a prefix for nested routes. Name is URL decoration only; id is the key. Call: {{template "list-path" (dict "name" $name "id" $id)}} */}} {{define "list-base"}}{{- $slug := kebabcase (trim .name) -}}{{- if eq $slug "" -}}{{- $slug = "list" -}}{{- end -}}/list/{{$slug}}/{{.id}}{{- end}} {{define "list-path"}}{{template "list-base" .}}/{{end}}
