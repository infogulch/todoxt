{{/* notify publishes a lightweight ping on the per-list bus topic so other tabs refresh. */}} {{define "notify"}} {{- $owner := or (.Req.Header.Get "X-Token-Subject") "anonymous"}} {{- $listId := .Req.PathValue "id"}} {{- if ne $listId ""}} {{- $_ := .Bus.Publish (printf "todos:%s:%s" $owner $listId) "updated"}} {{- end}} {{end}} {{/* ensure-lists creates a default "My Todos" list when the owner has none. Call only from the lists index (GET /) and POST /lists — not from list routes. */}} {{define "ensure-lists"}} {{- $owner := or (.Req.Header.Get "X-Token-Subject") "anonymous"}} {{- $n := .DB.QueryVal `SELECT COUNT(*) FROM todo_list WHERE owner_id = ?` $owner}} {{- if eq $n 0}} {{- $id := uuidv4}} {{- $_ := .DB.Exec `INSERT INTO todo_list (id, owner_id, name) VALUES (?, ?, ?)` $id $owner "My Todos"}} {{- end}} {{end}} {{/* require-list loads the path list for the current owner into .Vars "list" (map with id, name). Missing / not owned → 404 (replaces buffer). Call: {{template "require-list" .}} then $list := .Vars.Get "list" */}} {{define "require-list"}} {{- $owner := or (.Req.Header.Get "X-Token-Subject") "anonymous"}} {{- $id := .Req.PathValue "id"}} {{- $r := try .DB "QueryRow" ` SELECT id, name FROM todo_list WHERE id = ? AND owner_id = ?` $id $owner}} {{- if not $r.OK}} {{.Resp.RespondWith 404 (.X.Template "/shared/.404.html" .)}} {{- end}} {{- $_ := .Vars.Set "list" $r.Value -}} {{end}} {{/* list-path — canonical page URL for a list (trailing slash; file-based index route). list-base — same without trailing slash; use for DELETE and as a prefix for nested routes. Name is URL decoration only; id is the key. Call: {{template "list-path" (dict "name" $name "id" $id)}} */}} {{define "list-base"}}{{- $slug := kebabcase (trim .name) -}}{{- if eq $slug "" -}}{{- $slug = "list" -}}{{- end -}}/list/{{$slug}}/{{.id}}{{- end}} {{define "list-path"}}{{template "list-base" .}}/{{end}}
